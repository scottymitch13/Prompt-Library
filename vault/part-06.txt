const STORAGE_KEY = 'prompt-vault-v1';
let activeIndex = -1;
let pendingReset = null; // 'all' or index number

function loadStorage() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : {};
  } catch { return {}; }
}

function saveStorage(data) {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
}

// Returns the saved editable texts for a prompt, or defaults
function getPromptTexts(idx) {
  const store = loadStorage();
  return store[PROMPTS[idx].id] || null;
}

function savePromptTexts(idx, texts) {
  const store = loadStorage();
  store[PROMPTS[idx].id] = texts;
  saveStorage(store);
}

function clearPromptTexts(idx) {
  const store = loadStorage();
  delete store[PROMPTS[idx].id];
  saveStorage(store);
}

function clearAllTexts() {
  localStorage.removeItem(STORAGE_KEY);
}

// ── SIDEBAR NAV ----
function buildNav() {
  const nav = document.getElementById('promptNav');
  nav.innerHTML = '';
  let num = 0;
  GROUPS.forEach(g => {
    const header = document.createElement('div');
    header.className = 'nav-group';
    header.textContent = g.name;
    nav.appendChild(header);
    g.ids.forEach(id => {
      const i = PROMPTS.findIndex(p => p.id === id);
      if (i < 0) return;
      const p = PROMPTS[i];
      const item = document.createElement('div');
      item.className = 'nav-item';
      item.dataset.index = i;
      item.innerHTML = `<span class="nav-num">${String(num).padStart(2,'0')}</span><span class="nav-label">${p.title}</span>`;
      item.addEventListener('click', () => selectPrompt(i));
      nav.appendChild(item);
      num++;
    });
  });
}

function updateNavActive(idx) {
  document.querySelectorAll('.nav-item').forEach((el, i) => {
    el.classList.toggle('active', parseInt(el.dataset.index, 10) === idx);
  });
}

// ── SEARCH ----
document.getElementById('searchInput').addEventListener('input', function() {
  const q = this.value.toLowerCase().trim();
  document.querySelectorAll('.nav-item').forEach(el => {
    const p = PROMPTS[parseInt(el.dataset.index, 10)];
    const match = p.title.toLowerCase().includes(q) ||
                  p.tag.toLowerCase().includes(q) ||
                  p.subtitle.toLowerCase().includes(q) ||
                  (p.explain || '') + ' ' + (p.match || '').toLowerCase().includes(q);
    el.classList.toggle('hidden', q !== '' && !match);
  });
  document.querySelectorAll('.nav-group').forEach(h => {
    let sib = h.nextElementSibling;
    let any = false;
    while (sib && sib.classList.contains('nav-item')) {
      if (!sib.classList.contains('hidden')) { any = true; break; }
      sib = sib.nextElementSibling;
    }
    h.classList.toggle('hidden', !any);
  });
});

function selectPrompt(idx) {
  activeIndex = idx;
  const p = PROMPTS[idx];

  // Hide welcome, show prompt view
  document.getElementById('welcomeScreen').style.display = 'none';
  document.getElementById('promptView').style.display = 'block';
  document.getElementById('copyAllBtn').classList.remove('hidden');

  // Update top bar
  document.getElementById('topBarTitle').textContent = p.title;

  // Update meta
  const tagEl = document.getElementById('promptTag');
  tagEl.textContent = p.tag;
  tagEl.className = `prompt-tag ${p.tagClass}`;
  document.getElementById('promptTitle').textContent = p.title;
  document.getElementById('promptSubtitle').textContent = p.subtitle;

  const explainBox = document.getElementById('promptExplain');
  if (explainBox) {
    if (p.explain) {
      explainBox.innerHTML = '<span class="explain-label">In plain English</span>';
      const body = document.createElement('div');
      body.textContent = p.explain;
      explainBox.appendChild(body);
      explainBox.style.display = 'block';
    } else {
      explainBox.style.display = 'none';
    }
  }

  // Load saved texts
  const saved = getPromptTexts(idx);
  let editableIdx = 0;
  const editableDefaults = p.blocks
    .filter(b => b.type === 'editable')
    .map(b => b.text);

  const currentTexts = saved || editableDefaults;

  // Render blocks
  const area = document.getElementById('editorArea');
  area.innerHTML = '';

  let openGroup = null; // tracks current "group" div

  p.blocks.forEach((block, bi) => {
    if (block.type === 'divider') {
      const d = document.createElement('div');
      d.className = 'section-divider';
      area.appendChild(d);
      openGroup = null;
      return;
    }

    if (block.type === 'static') {
      const el = document.createElement('div');
      el.className = 'block-static';
      el.textContent = block.text;
      area.appendChild(el);
      openGroup = null;
      return;
    }

    if (block.type === 'step') {
      const el = document.createElement('div');
      el.className = 'step-header';
      el.innerHTML = `<span class="step-label">${block.label}</span>${block.title ? `<span class="step-title">${block.title}</span>` : ''}`;
      area.appendChild(el);
      openGroup = null;
      return;
    }

    if (block.type === 'editable') {
      const savedText = currentTexts[editableIdx] !== undefined
        ? currentTexts[editableIdx]
        : block.text;
      const isPlaceholder = savedText === block.text && block.placeholder;

      const wrap = document.createElement('div');
      wrap.className = 'block-editable';

      const ta = document.createElement('textarea');
      ta.value = savedText;
      ta.dataset.editableIdx = editableIdx;
      ta.dataset.blockIdx = bi;
      ta.spellcheck = false;

      if (isPlaceholder) ta.classList.add('is-placeholder');

      // Auto-resize
      ta.addEventListener('input', () => {
        autoResize(ta);
        // Remove placeholder style once user types
        ta.classList.remove('is-placeholder');
        persistCurrentPrompt();
      });

      ta.addEventListener('focus', () => {
        if (ta.classList.contains('is-placeholder')) {
          // Don't auto-clear — let user edit in place
        }
      });

      const tag = document.createElement('span');
      tag.className = 'placeholder-tag';
      tag.textContent = block.placeholder ? 'Fill in' : 'Editable';

      wrap.appendChild(ta);
      wrap.appendChild(tag);
      area.appendChild(wrap);

      // Defer resize until rendered
      requestAnimationFrame(() => autoResize(ta));

      editableIdx++;
    }
  });

  updateNavActive(idx);
  showSaveIndicator('');
}

function autoResize(ta) {
  ta.style.height = 'auto';
  ta.style.height = (ta.scrollHeight) + 'px';
}

// ── PERSIST ----
function persistCurrentPrompt() {
  if (activeIndex < 0) return;
  const texts = [];
  document.querySelectorAll('#editorArea .block-editable textarea').forEach(ta => {
    texts.push(ta.value);
  });
  savePromptTexts(activeIndex, texts);
  showSaveIndicator('Saved');
}

function showSaveIndicator(msg) {
  const el = document.getElementById('saveIndicator');
  el.textContent = msg;
  el.classList.toggle('visible', msg !== '');
  if (msg) {
    clearTimeout(el._t);
    el._t = setTimeout(() => { el.classList.remove('visible'); }, 2000);
  }
}

// ── COPY PROMPT ----
function buildFullPromptText() {
  if (activeIndex < 0) return '';
  const p = PROMPTS[activeIndex];
  const textareas = document.querySelectorAll('#editorArea .block-editable textarea');
  let taIdx = 0;
  const parts = [];

  p.blocks.forEach(block => {
    if (block.type === 'static') {
      parts.push(block.text);
    } else if (block.type === 'step') {
      parts.push(`\n${block.label}${block.title}`);
    } else if (block.type === 'editable') {
      const ta = textareas[taIdx];
      if (ta) parts.push(ta.value);
      taIdx++;
    } else if (block.type === 'divider') {
      parts.push('');
    }
  });

  return parts.join('\n\n').replace(/\n{3,}/g, '\n\n').trim();
}

function copyToClipboard(text) {
  if (navigator.clipboard && window.isSecureContext) {
    return navigator.clipboard.writeText(text);
  } else {
    const ta = document.createElement('textarea');
    ta.value = text;
    ta.style.position = 'fixed';
    ta.style.opacity = '0';
    document.body.appendChild(ta);
    ta.focus();
    ta.select();
    document.execCommand('copy');
    document.body.removeChild(ta);
    return Promise.resolve();
  }
}

function showToast(msg) {
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.classList.add('show');
  clearTimeout(t._t);
  t._t = setTimeout(() => t.classList.remove('show'), 2200);
}

(function() {
  var STOP = new Set(['i','a','an','the','to','my','me','of','and','for','it','this','that','is','are','on','in','do','can','you','your','want','need','help','with','make','get','have','be','what','how','so','we','at','or','from','by','up','out','into','about','was','will','would','like','please','some','something','they','them','when','then','than','just','really','use','using','one','way']);
  var ID2GROUP = {};
  GROUPS.forEach(function(g) { g.ids.forEach(function(id) { ID2GROUP[id] = g.name; }); });
  var INDEX = PROMPTS.map(function(p, i) {
    return {
      i: i,
      strong: ((p.match || '') + ' ' + p.title + ' ' + (p.tag || '')).toLowerCase(),
      weak: ((p.subtitle || '') + ' ' + (p.explain || '')).toLowerCase()
    };
  });
  function esc(t) { return t.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'); }
  function tokens(q) {
    return q.replace(/[^a-z0-9\s'-]/g, ' ').split(/\s+/).filter(function(t) { return t.length >= 3 && !STOP.has(t); });
  }
  function score(e, toks, raw) {
    var sc = 0;
    toks.forEach(function(t) {
      var re = new RegExp('\\b' + esc(t));
      if (re.test(e.strong)) sc += 6;
      else if (re.test(e.weak)) sc += 2;
    });
    if (raw.length >= 8) {
      if (e.strong.indexOf(raw) !== -1) sc += 10;
      else if (e.weak.indexOf(raw) !== -1) sc += 5;
    }
    return sc;
  }
  function firstSentence(t) {
    var m = (t || '').match(/^.*?[.!?](\s|$)/);
    return m ? m[0].trim() : (t || '');
  }
  var askInput = document.getElementById('askInput');
  var askResults = document.getElementById('askResults');
  var currentTop = null;
  function render(ranked) {
    askResults.innerHTML = '';
    if (!ranked.length) {
      currentTop = null;
      var d = document.createElement('div');
      d.className = 'ask-empty';
      d.textContent = 'No obvious match. Try different words, or browse the groups below.';
      askResults.appendChild(d);
      return;
    }
    currentTop = ranked[0].e.i;
    ranked.forEach(function(r, k) {
      var p = PROMPTS[r.e.i];
